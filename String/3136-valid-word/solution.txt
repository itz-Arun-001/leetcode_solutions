class Solution {
public:
    bool isValid(string word) {

        int n=word.size(),a=0,b=0;
        for(char c:word)
        {
            if(isalnum(c))
            {
                if(c>='0'&&c<='9')
                    continue;
                else if(c=='a'||c=='e'||c=='i'||c=='o'||c=='u'||c=='A'||c=='E'||c=='I'||c=='O'||c=='U')
                    a++;
                else b++;
            }
            else return false;
        }
        return (n>=3&&a>0&&b>0)?true:false;
        
    }
};
