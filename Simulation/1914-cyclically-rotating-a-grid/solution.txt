class Solution {
public:
    vector<vector<int>> rotateGrid(vector<vector<int>>& grid, int k) {
        int m=grid.size();
        int n=grid[0].size();
        int c=min(n,m)/2;
        for(int cnt=0;cnt<c;cnt++)
        {
            vector<int>temp;
            int top=cnt,bot=m-cnt-1,left=cnt,right=n-cnt-1;
            for(int i=left;i<=right;i++)
            {
                temp.push_back(grid[top][i]);
            }
            for(int i=top+1;i<=bot;i++)
                temp.push_back(grid[i][right]);
            if(bot>top)
            {
                for(int i=right-1;i>=left;i--)
                    temp.push_back(grid[bot][i]);
            }
            if(left<right)
            {
                for(int i=bot-1;i>top;i--)
                {
                    temp.push_back(grid[i][left]);
                }
            }
            int x=k%temp.size();
            reverse(temp.begin(),temp.begin()+x);
            reverse(temp.begin()+x,temp.end());
            reverse(temp.begin(),temp.end());
            int idx=0;
            for (int j = left; j <= right; j++)
                grid[top][j] = temp[idx++];
            for (int i = top + 1; i <= bot; i++)
                grid[i][right] = temp[idx++];
            if (top < bot) {
                for (int j = right - 1; j >= left; j--)
                    grid[bot][j] = temp[idx++];
            }
            if (left < right) {
                for (int i = bot - 1; i > top; i--)
                    grid[i][left] = temp[idx++];
            }
        }
        return grid;
    }
};
