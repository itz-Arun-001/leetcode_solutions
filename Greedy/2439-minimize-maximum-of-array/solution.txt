class Solution {
public:
    int check(vector<int> nums, int mid) {
        long long sum = 0;
        for (int i = 0; i < nums.size(); i++){
            sum+=nums[i];
            if(sum>1LL*(i+1)*mid)
               return INT_MAX;
        }
        return mid;
    }
    int minimizeArrayValue(vector<int>& nums) {
        int low = 0;
        int high = *max_element(nums.begin(), nums.end());
        int ans = INT_MAX;

        while (low <= high) {
            int mid = low + (high - low) / 2;
            int val = check(nums, mid);
            if (ans > val) {
                ans = val;
                high = mid - 1;
            } else
                low = mid + 1;
        }
        return ans;
    }
};
