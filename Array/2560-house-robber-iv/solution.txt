class Solution {
public:
 bool check(vector<int>& nums, int mid, int k) {
        int cnt = 0;
        for (int i = 0; i < nums.size(); i++) {
            if (nums[i] <= mid) {
                cnt++;   
                i++;     
            }
        }
        return cnt >= k;
    }
    int minCapability(vector<int>& nums, int k) {
        int mn=*min_element(nums.begin(),nums.end());
        int mx=*max_element(nums.begin(),nums.end());
        int l=mn,h=mx;
        int ans=INT_MAX;
        while(l<=h)
        {
            int m=l+(h-l)/2;
            if(check(nums,m,k))
            {
                ans=m;
                h=m-1;

            }
            else l=m+1;

        }
        return ans;
    }
};
